Oscilador Cuantico 3D

Esta simulacion representa un oscilador armonico anisotropo, donde la fuerza restauradora no es igual en los tres ejes: cada direccion espacial (x, y, z) tiene su propia frecuencia y por eso la funcion de onda adopta estructuras diferentes segun el eje, con nodos y regiones de mayor probabilidad caracteristicos. La escena muestra una superposicion de dos estados cuanticos y su interferencia temporal: cada estado evoluciona con su fase, y la diferencia de energia entre ambos produce oscilaciones en el patron de probabilidad, haciendo que la intensidad espacial se redistribuya periodicamente en el tiempo.

Si nunca viste fisica cuantica, piensa esto:

No estamos viendo una bolita moviendose por una ruta fija. Estamos viendo un "mapa de posibilidad": en que lugares del espacio es mas probable encontrar la particula si la midieras.

Cada punto de la nube es una muestra de ese mapa. Muchas muestras juntas en una zona significan "aqui suele aparecer"; pocas muestras significan "aqui casi no aparece".

La nube cambia con el tiempo porque se mezclan dos estados cuanticos. Esa mezcla genera un pulso visual: las zonas intensas suben y bajan periodicamente.

Explicacion de Codigo

// ================= INCLUSIONES =================

    1   #include <cmath>                           // Funciones matemáticas del oscilador
    2   #include <random>                          // Generación aleatoria
    3   #include <vector>                           // Almacenamiento de partículas
    4   #include <emscripten.h>                     // Bucle WebAssembly en navegador
    5   #include <GLES2/gl2.h>                      // Renderizado WebGL
    6   #include <GLFW/glfw3.h>                     // Ventana y contexto OpenGL

    // ================= PARÁMETROS FÍSICOS =================

    10  const double X_MAX = 5.0;                    // Tamaño del dominio espacial
    11  const int NUM_PARTICLES = 150000;             // Cantidad de puntos de la nube
    12  const double REJECTION_THRESHOLD = 0.0001;    // Umbral del muestreo por rechazo
    13  std::vector<float> voxel_data;               // [x, y, z, ψA, ψB]
    14  std::mt19937 gen(std::random_device{}());     // Motor aleatorio
    15  std::uniform_real_distribution<double> dist_space(-X_MAX, X_MAX); // Posiciones uniformes
    16  float time_val = 0.0f;                        // Tiempo de evolución
    17  float delta_e = 0.0f;                         // Diferencia de energía entre estados
    18  GLuint program, vbo;                          // Programa shader y búfer

    // ================= FUNCIONES DE ONDA =================

    24  double get_normalization(int n, double omega) {
    25      return std::pow(omega / M_PI, 0.25) /
    26             std::sqrt(std::pow(2.0, n) * std::tgamma(n + 1.0)); // Normalización
    27  }
    28  double psi_1d(int n, double x, double omega) {
    29      double q = x * std::sqrt(omega);           // Coordenada adimensional
    30      return get_normalization(n, omega) *
    31             std::hermite(n, q) * std::exp(-0.5 * q * q); // ψ_n(x)
    32  }

    // ================= MUESTREO POR RECHAZO =================

    40  extern "C" EMSCRIPTEN_KEEPALIVE void update_quantum_state(
    41      int nxA, int nyA, int nzA, int nxB, int nyB, int nzB,
    42      float wx, float wy, float wz) {
    43      double omega_x = std::max(0.05, static_cast<double>(wx));
    44      double omega_y = std::max(0.05, static_cast<double>(wy));
    45      double omega_z = std::max(0.05, static_cast<double>(wz));
    46      delta_e = static_cast<float>(
    47          (omega_x * (nxB + 0.5) + omega_y * (nyB + 0.5) + omega_z * (nzB + 0.5)) -
    48          (omega_x * (nxA + 0.5) + omega_y * (nyA + 0.5) + omega_z * (nzA + 0.5))
    49      ); // ΔE
    50      voxel_data.clear();
    51      voxel_data.reserve(NUM_PARTICLES * 5);
    52      int accepted = 0;
    53      while (accepted < NUM_PARTICLES) {
    54          double x = dist_space(gen), y = dist_space(gen), z = dist_space(gen); // Muestreo uniforme
    55          double psiA = psi_1d(nxA, x, omega_x) *
    56                        psi_1d(nyA, y, omega_y) *
    57                        psi_1d(nzA, z, omega_z); // ψ_A
    58          double psiB = psi_1d(nxB, x, omega_x) *
    59                        psi_1d(nyB, y, omega_y) *
    60                        psi_1d(nzB, z, omega_z); // ψ_B
    61          if (psiA * psiA + psiB * psiB > REJECTION_THRESHOLD) { // Aceptación por probabilidad
    62              voxel_data.push_back(static_cast<float>(x));
    63              voxel_data.push_back(static_cast<float>(y));
    64              voxel_data.push_back(static_cast<float>(z));
    65              voxel_data.push_back(static_cast<float>(psiA));
    66              voxel_data.push_back(static_cast<float>(psiB));
    67              ++accepted;
    68          }
    69      }
    70      glBindBuffer(GL_ARRAY_BUFFER, vbo); // Enviar datos a la GPU
    71      glBufferData(GL_ARRAY_BUFFER, voxel_data.size() * sizeof(float), voxel_data.data(), GL_STATIC_DRAW);
    72  }

    // ================= SHADERS =================

    70  const char* vertex_shader_src = R"(
    71      attribute vec3 a_pos;               // Posición de la partícula
    72      attribute float a_psiA;             // Amplitud del estado A
    73      attribute float a_psiB;             // Amplitud del estado B
    74      varying vec3 v_pos;
    75      varying float v_psiA;
    76      varying float v_psiB;
    77      uniform float u_yaw;                // Rotación horizontal
    78      uniform float u_pitch;              // Rotación vertical
    79      uniform float u_zoom;               // Escala de vista
    80      void main() {
    81          mat3 rotY = mat3(cos(u_yaw), 0.0, -sin(u_yaw),
    82                           0.0, 1.0, 0.0,
    83                           sin(u_yaw), 0.0, cos(u_yaw)); // Giro Y
    84          mat3 rotX = mat3(1.0, 0.0, 0.0,
    85                           0.0, cos(u_pitch), sin(u_pitch),
    86                           0.0, -sin(u_pitch), cos(u_pitch)); // Giro X
    87          v_pos = a_pos; v_psiA = a_psiA; v_psiB = a_psiB;
    88          gl_Position = vec4(rotY * rotX * (a_pos * 0.2 * u_zoom), 1.0); // Posición final
    85          gl_PointSize = 3.0;
    86      }
    87  )";

    89  const char* fragment_shader_src = R"(
    90      precision mediump float;
    91      varying vec3 v_pos;                 // Posición del punto
    92      varying float v_psiA;               // Amplitud A
    93      varying float v_psiB;               // Amplitud B
    94      uniform float u_color_scale;        // Escala de color
    95      uniform float u_time;               // Tiempo
    96      uniform float u_delta_e;            // Diferencia de energía
    97      void main() {
    98          float p = 0.5 * v_psiA * v_psiA + 0.5 * v_psiB * v_psiB +
    99                    v_psiA * v_psiB * cos(u_delta_e * u_time); // Interferencia
    100         float mapped = clamp(
    101             log(1.0 + max(p, 0.0) * 30.0 * max(0.1, u_color_scale) * 25.0) /
    102             log(1.0 + max(0.1, u_color_scale) * 25.0), 0.0, 1.0
    103         ); // Escala logarítmica
    104         vec3 c = mix(
    105             mix(
    106                 mix(vec3(0.0), vec3(0.5, 0.0, 0.5), smoothstep(0.0, 0.20, mapped)),
    107                 vec3(1.0, 1.0, 0.0),
    108                 smoothstep(0.20, 0.72, mapped)
    109             ),
    110             vec3(1.0),
    111             smoothstep(0.72, 1.0, mapped)
    112         );
    113         gl_FragColor = vec4(c, 0.75);
    102     }
    103 )";

    // ================= BUCLE Y ARRANQUE =================

    110 void render_loop() {
    111     time_val += 0.016f;                         // Avance temporal
    112     glClear(GL_COLOR_BUFFER_BIT);               // Limpiar pantalla
    113     glUseProgram(program);                      // Activar shader
    114     glUniform1f(glGetUniformLocation(program, "u_time"), time_val);
    115     glUniform1f(glGetUniformLocation(program, "u_delta_e"), delta_e);
    116     glBindBuffer(GL_ARRAY_BUFFER, vbo);         // Usar búfer de partículas
    117     glDrawArrays(GL_POINTS, 0, static_cast<GLsizei>(voxel_data.size() / 5)); // Dibujar nube
    118     glfwSwapBuffers(window);                    // Mostrar fotograma
    119     glfwPollEvents();                           // Procesar eventos
    120 }

    122 int main() {
    123     glfwInit();                                  // Inicializar ventana
    124     window = glfwCreateWindow(800, 600, "Oscilador", nullptr, nullptr); // Crear ventana
    125     glfwMakeContextCurrent(window);              // Activar contexto
    126     glEnable(GL_BLEND);                          // Habilitar alfa
    127     glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA); // Modo de mezcla
    128     program = glCreateProgram();                 // Crear programa shader
    129     glAttachShader(program, compile_shader(GL_VERTEX_SHADER, vertex_shader_src));
    130     glAttachShader(program, compile_shader(GL_FRAGMENT_SHADER, fragment_shader_src));
    131     glLinkProgram(program);
    132     glGenBuffers(1, &vbo);                       // Crear búfer GPU
    133     update_quantum_state(0, 0, 0, 1, 0, 0, 1.0f, 1.0f, 1.0f); // Estado inicial
    134     emscripten_set_main_loop(render_loop, 0, 1); // Iniciar animación
    135     return 0;
    136 }
